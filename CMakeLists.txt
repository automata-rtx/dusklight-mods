cmake_minimum_required(VERSION 3.26)
project(dusklight_mods CXX)

# Superproject for our Dusklight graphics mods, built on the official Dusklight mod template
# (https://github.com/TwilitRealm/mod-template): the pinned Dusklight source is fetched by
# DUSKLIGHT_VERSION into ./dusklight (cmake/FetchDusklight.cmake), each mod is packaged as a .dusk
# into ${CMAKE_BINARY_DIR}/mods, and the SDK auto-downloads the per-arch link stub it needs. Only
# the mod sources compile here — the game and its renderer are never built.
#
# The platform is the SCENE NORMAL BUFFER build: automata-rtx/dusklight-ao's
# claude/dusklight-thin-gbuffer-normals-l4l9dc, published as the platform-normals-test prerelease.
# It is upstream Dusklight c880d46f plus the GfxService 1.3 scene-normal snapshot, over upstream
# aurora 0f15fb6 (upstream-sized streaming buffers) plus the optional normal attachment.
#
# Two reasons this is the pin rather than upstream:
#
#   1. ABI. Upstream bumped the GAME SERVICE MAJOR VERSION at 008a18c1. Mods built against the
#      older 0fc05028 SDK are refused outright by this host - which is why every mod but SMAA
#      (whose imports do not reach it) failed to load rather than merely misbehaving. 0fc05028 is
#      also where this repo's main branch sat while the fork was retired, below.
#   2. Authored normals. GfxService 1.3 hands mods the game's own interpolated vertex normal.
#      See docs/authored_normals.md.
#
# THE FORK DELTA IS ONE SERVICE CALL. Everything else the mods use is upstream: GfxService 1.2
# (GfxRenderTargetLayout, get_scene_target_layout, gfx_init_color_target_states, GfxDrawContext
# ::layout) landed upstream in #2305, so the hand-rolled GfxPassTargets vocabulary our fork used to
# carry was deleted rather than merged. What remains fork-local is 1.3's
#
#     ModResult get_scene_normals(ModContext*, GfxSceneNormals*)
#
# plus aurora's optional normal attachment behind it. The HOST snapshots that attachment once per
# frame, right after the opaque lists and before any SCENE_AFTER_OPAQUE hook, and hands the same
# texture to every mod that asks - so the pass break and the copy are paid once however many mods
# read normals. It is ALWAYS ON: there is no video setting and no restart, and it is unavailable
# only on the compatibility renderers (D3D11 / OpenGL ES), which cannot carry the attachment.
#
# This supersedes an earlier shape of the same feature that added a per-mod resolve
# (GfxResolveDesc::normal -> GfxResolvedTargets::normal) and, before that, a mod-side normal
# PROVIDER. Both are retired: a mod now asks the service directly and needs no other mod.
#
# WE RETIRED AN EARLIER FORK OVER AN OFFSET COLLISION, AND THAT LESSON STILL STANDS. That fork
# appended GfxDeviceInfo::normal_format at offset 40; upstream independently appended WGPUInstance
# at the same offset, so a fork-built mod read a live pointer as a texture format, concluded the
# g-buffer existed, declared a second colour target against a one-attachment scene pass, and had
# every composite rejected - the mods loaded and did nothing. The tree retreated to upstream
# 0fc05028 for that reason. It cannot recur HERE because the delta adds no field to any existing
# struct: get_scene_normals is APPENDED TO THE SERVICE VTABLE behind a minor-version bump, and a
# host that predates it simply fails the required-import check at load, loudly. That is the shape
# to prefer if the fork ever grows again. See docs/normal_buffer_portability.md.
#
# DUSKLIGHT_SDK_STUB_URL must track DUSKLIGHT_VERSION. Upstream's own stub release is public and
# version-INDEPENDENT (one fixed `sdk` tag, generated by symgen) and carries the whole CI asset
# matrix, so tracking upstream needs no stub knob at all - but it does not carry THIS build's
# symbols, so a fork pin must name the per-arch stubs from the same prerelease as the game build.
# (Linux needs no stub either way: game symbols resolve at load via --allow-shlib-undefined.) The
# `platform-normals-test` tag is REPUBLISHED on every push to the platform branch, so the URL is
# stable across re-pins while its assets are not - a stale local build/ directory can therefore hold
# stubs from the previous build. DUSKLIGHT_AURORA_VERSION stays unset: the recorded extern/aurora
# pin (299ed44) is a live branch tip on automata-rtx/aurora-ao and resolves on its own.
#
# To move to a newer base game, bump DUSKLIGHT_VERSION and the matching stub URL, and reconfigure.
# Keep the pin matched to the game build actually being run — the game-linked mods resolve hook
# targets by symbol at load, and a mod built against a different base can fail to load outright.
if (NOT DUSKLIGHT_VERSION)
    set(DUSKLIGHT_VERSION "af87a9ec2c7617b1c8b6dec9799fbf3d83830764")
endif ()

if (NOT DEFINED DUSKLIGHT_REPOSITORY)
    set(DUSKLIGHT_REPOSITORY "https://github.com/automata-rtx/dusklight-ao.git"
            CACHE STRING "Dusklight git repository to fetch from")
endif ()

if (NOT DEFINED DUSKLIGHT_SDK_STUB_URL)
    set(DUSKLIGHT_SDK_STUB_URL
            "https://github.com/automata-rtx/dusklight-ao/releases/download/platform-normals-test"
            CACHE STRING "Base URL for game link stubs downloaded by out-of-tree mod builds")
endif ()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchDusklight.cmake")

add_subdirectory("${DUSKLIGHT_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)

# THREE MODS BUILD RIGHT NOW — VBAO, Deferred Fog, SMAA. Deliberate scope control: a test drop is
# exactly these three, not a mixed bag of mods at different stages. CI globs build/mods/*.dusk, so
# gating the add_subdirectory calls here is all it takes.
#
# Order here is alphabetical-ish and does not matter: no mod in this set imports another. The
# AO-under-fog ordering that Deferred Fog exists for is guaranteed at RUNTIME by stage separation
# (VBAO composites at SCENE_AFTER_OPAQUE, the fog quad draws at FRAME_BEFORE_HUD), not by build or
# load order.
add_subdirectory(mods/deferred_fog)
add_subdirectory(mods/vbao)
add_subdirectory(mods/smaa)
