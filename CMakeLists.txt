cmake_minimum_required(VERSION 3.26)
project(dusklight_mods CXX)

# Superproject for our Dusklight graphics mods, built on the official Dusklight mod template
# (https://github.com/TwilitRealm/mod-template): the pinned Dusklight source is fetched by
# DUSKLIGHT_VERSION into ./dusklight (cmake/FetchDusklight.cmake), each mod is packaged as a .dusk
# into ${CMAKE_BINARY_DIR}/mods, and the SDK auto-downloads the per-arch link stub it needs. Only
# the mod sources compile here — the game and its renderer are never built.
#
# This tracks UPSTREAM Dusklight directly — there is no fork delta left. DUSKLIGHT_REPOSITORY,
# DUSKLIGHT_SDK_STUB_URL and DUSKLIGHT_AURORA_VERSION are all deliberately unset here so the
# template's own defaults apply: the source comes from TwilitRealm/dusklight, and the SDK downloads
# its per-arch link stub from encounter/dusklight releases/download/sdk. That stub release is
# public and version-INDEPENDENT (one fixed `sdk` tag, generated by symgen), and it carries every
# asset our CI matrix needs — windows-amd64.lib, windows-arm64.lib, stub-macos-arm64,
# stub-macos-x86_64, stub-android-aarch64.so. Linux needs no stub at all: game symbols resolve at
# load via --allow-shlib-undefined. All three knobs remain settable from the command line.
#
# We previously pinned our own fork (automata-rtx/dusklight-ao) for the thin g-buffer authored
# normals. Upstream has no normal buffer of any kind and appended WGPUInstance/WGPUAdapter to
# GfxDeviceInfo in the very slot our fork appended normal_format, so a fork-built mod read a live
# pointer as a texture format, concluded the g-buffer existed, and declared a second colour target
# against a one-attachment scene pass — every composite was rejected and the mods did nothing.
# Mod code needs no #ifdefs for this: common/gfx_normal_compat.h detects the fields at compile time
# and degrades to "no normal buffer". See docs/normal_buffer_portability.md.
#
# To move to a newer base game, bump DUSKLIGHT_VERSION (one line) and reconfigure. Keep it matched
# to the game build actually being run — the game-linked mods resolve hook targets by symbol at
# load, and a mod built against a different base can fail to load outright.
if (NOT DUSKLIGHT_VERSION)
    set(DUSKLIGHT_VERSION "0fc05028ccfe809c569b1b84c0bb87f382b0bf34")
endif ()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchDusklight.cmake")

add_subdirectory("${DUSKLIGHT_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)

# Graphics Hub exports the depth-to-normal service and Celestial Orbit the sun/moon orbit service;
# add both before their consumers (ssilvb, realtime_sun_shadows), which add their include/ for the
# shared service headers.
add_subdirectory(mods/graphics_hub)
add_subdirectory(mods/celestial_orbit)
add_subdirectory(mods/vbao)
add_subdirectory(mods/ssilvb)
add_subdirectory(mods/realtime_sun_shadows)
add_subdirectory(mods/effect_remover)
add_subdirectory(mods/smaa)
add_subdirectory(mods/lantern_position)
