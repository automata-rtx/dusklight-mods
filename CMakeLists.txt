cmake_minimum_required(VERSION 3.26)
project(dusklight_mods CXX)

# Superproject for our Dusklight graphics mods, built on the official Dusklight mod template
# (https://github.com/TwilitRealm/mod-template): the pinned Dusklight source is fetched by
# DUSKLIGHT_VERSION into ./dusklight (cmake/FetchDusklight.cmake), each mod is packaged as a .dusk
# into ${CMAKE_BINARY_DIR}/mods, and the SDK auto-downloads the per-arch link stub it needs. Only
# the mod sources compile here — the game and its renderer are never built.
#
# THE PLATFORM IS UPSTREAM DUSKLIGHT AGAIN. The fork this tree ran on for months
# (automata-rtx/dusklight-ao) existed for exactly one reason — it was the only build that handed
# mods the game's authored vertex normals — and upstream now ships that itself. GfxService 1.3 and
# GameService 2.0 are upstream, the normal attachment is in upstream aurora (encounter/aurora, the
# recorded extern/aurora pin), and nothing in this repo needs the fork any more. Both fork knobs
# below are therefore gone, not merely defaulted: DUSKLIGHT_REPOSITORY is upstream, and there is no
# DUSKLIGHT_SDK_STUB_URL override because upstream's stub release is version-INDEPENDENT (one fixed
# `sdk` tag generated by symgen) and the SDK's own default already points at it. DUSKLIGHT_AURORA_
# VERSION is gone too — it only ever existed because a force-pushed fork branch could leave the
# submodule pin dangling.
#
# UPSTREAM'S NORMAL API IS NOT THE ONE OUR FORK HAD. Same service version, different shape, and the
# difference is binary, so read this before assuming a mod written against the fork still works:
#
#   fork (retired)      svc_gfx->get_scene_normals(ctx, &GfxSceneNormals)  — a separate vtable call,
#                       host-snapshotted once per frame whether or not anyone asked. `normal` on
#                       GfxResolveDesc lived in the struct's TAIL PADDING, so sizeof was unchanged.
#   upstream (now)      GfxResolveDesc::normal -> GfxResolvedTargets::normal, through resolve_pass.
#                       There is NO get_scene_normals in the upstream vtable at all. `normal` is a
#                       uint32_t APPENDED to GfxResolveDesc — "not a bool to avoid using the
#                       previous padding", in upstream's own comment — so sizeof GREW, 8 -> 12.
#
# That last point is why this is a real re-platform and not a cosmetic one. The host reads
# `desc->struct_size >= sizeof(GfxResolveDesc) ? desc->normal : 0` (src/dusk/mods/svc/gfx.cpp), so a
# mod built against the fork's SDK does not fail loudly on upstream — it silently gets NO normals,
# and a normal-consuming mod then disables itself. Rebuild every normal consumer against this pin.
#
# Mods reach both fields through common/gfx_normal_compat.h rather than touching them directly.
# That shim was written for this exact API when the fork still carried it, detects the fields by
# member name, and needed no change to work against upstream — which is the whole reason it exists.
#
# ONE BEHAVIOURAL CHANGE COMES WITH THE NEW API: the snapshot LATCHES. The first resolve that asks
# for normals enables the attachment for the NEXT frame and returns null for this one (upstream's
# mods/ao_mod says so in as many words, and its stage callback just returns). A consumer must
# therefore treat an early null as "not yet", not as "this device cannot do it" — see VBAO's
# g_normalWaitFrames. The old fork API had no such latch, so this is new behaviour to honour.
#
# The latch has a second consequence that no compile-time shim can absorb: the SCENE PASS GAINS A
# SECOND COLOUR ATTACHMENT when it fires, so a pipeline built at init against the one-attachment
# pass is rejected from then on, silently — the composite simply stops appearing. Scene-pass
# pipelines must be built lazily and rebuilt when GfxDrawContext::layout.key changes; see
# common/gfx_scene_pass.h and the ensure_*_pipelines() helpers in mods/vbao and mods/smaa.
#
# AND THE BUFFER WILL NOT EXIST AT ALL WITH MSAA ON. aurora only creates it when msaaSamples == 1
# (lib/webgpu/gpu.cpp enable_normal_buffer()), and resolve_pass does not even record the request
# otherwise. Dusklight never assigns AuroraConfig::msaa and aurora normalises 0 -> 1, so it is 1 in
# practice — but that is a default, not a guarantee, and both mods name MSAA in their log line when
# the live sample count says that is why they have no normals.
#
# To move to a newer base game, bump DUSKLIGHT_VERSION and reconfigure. Keep the pin matched to the
# game build actually being run — the game-linked mods resolve hook targets by symbol at load, and a
# mod built against a different base can fail to load outright.
if (NOT DUSKLIGHT_VERSION)
    set(DUSKLIGHT_VERSION "c83ce894e66d6d820a517068802c6d3179c9e08e")
endif ()

if (NOT DEFINED DUSKLIGHT_REPOSITORY)
    set(DUSKLIGHT_REPOSITORY "https://github.com/TwilitRealm/dusklight.git"
            CACHE STRING "Dusklight git repository to fetch from")
endif ()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchDusklight.cmake")

add_subdirectory("${DUSKLIGHT_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)

# WHICH MODS BUILD. Deliberate scope control: a test drop is a known set, not a mixed bag of mods at
# different stages. CI globs build/mods/*.dusk, so gating the add_subdirectory calls here is all it
# takes.
#
# Order does not matter: no mod in this set imports another.
add_subdirectory(mods/vbao)
add_subdirectory(mods/smaa)
